cmake_minimum_required(VERSION 3.25)
project(MsimeWebviewContract LANGUAGES CXX)
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost REQUIRED COMPONENTS json)
enable_testing()
add_executable(test_webview_contract test.cpp)
target_compile_features(test_webview_contract PRIVATE cxx_std_17)
target_link_libraries(test_webview_contract PRIVATE Boost::json)
add_test(NAME webview_contract COMMAND test_webview_contract "${CMAKE_CURRENT_SOURCE_DIR}/fixtures.json")
